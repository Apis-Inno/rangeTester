// src/rangeTest.cpp — see firmware file in project
SYSTEM_MODE(SEMI_AUTOMATIC);
SYSTEM_THREAD(ENABLED);

static bool atCb(int,const char*b,int l,void*){
  if(b&&l>0)Serial.write((uint8_t*)b,l);
  return true;
}
void setup(){ Serial.begin(115200); }
void loop(){
  if(Serial.available()){
    String s=Serial.readStringUntil('\n');
    s.trim();
    if(s.length())
      Cellular.command(atCb,NULL,30000,"%s\r\n",s.c_str());
  }
}
